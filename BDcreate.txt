-- =============================================
-- 1. Создание базы данных и выбор её
-- =============================================
DROP DATABASE IF EXISTS library_db;
CREATE DATABASE library_db
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;
USE library_db;

-- =============================================
-- 2. Таблица пользователей (сотрудников)
-- =============================================
CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) UNIQUE NOT NULL,
    password VARCHAR(100) NOT NULL,
    role VARCHAR(30) NOT NULL   -- 'ADMIN', 'LIBRARIAN'
);

-- =============================================
-- 3. Таблица книг
-- =============================================
CREATE TABLE books (
    id INT PRIMARY KEY AUTO_INCREMENT,
    title VARCHAR(200) NOT NULL,
    author VARCHAR(100),
    isbn VARCHAR(20),
    year INT,
    publisher VARCHAR(100),
    quantity INT NOT NULL DEFAULT 0,
    available INT NOT NULL DEFAULT 0
);

-- =============================================
-- 4. Таблица читателей
-- =============================================
CREATE TABLE readers (
    id INT PRIMARY KEY AUTO_INCREMENT,
    card_number VARCHAR(20) UNIQUE NOT NULL,
    full_name VARCHAR(150) NOT NULL,
    birth_date DATE,
    phone VARCHAR(20),
    address VARCHAR(200),
    registration_date DATE NOT NULL,
    active TINYINT(1) DEFAULT 1
);

-- =============================================
-- 5. Таблица выдач книг
-- =============================================
CREATE TABLE loans (
    id INT PRIMARY KEY AUTO_INCREMENT,
    book_id INT NOT NULL,
    reader_id INT NOT NULL,
    loan_date DATE NOT NULL,
    due_date DATE NOT NULL,
    return_date DATE,
    status VARCHAR(20) DEFAULT 'ACTIVE',  -- 'ACTIVE', 'RETURNED', 'OVERDUE'
    FOREIGN KEY (book_id) REFERENCES books(id),
    FOREIGN KEY (reader_id) REFERENCES readers(id)
);

-- =============================================
-- 6. Таблица заявок на закупку
-- =============================================
CREATE TABLE purchase_requests (
    id INT PRIMARY KEY AUTO_INCREMENT,
    request_date DATE NOT NULL,
    status VARCHAR(20) DEFAULT 'NEW',   -- 'NEW', 'RECEIVED'
    total_items INT DEFAULT 0
);

-- =============================================
-- 7. Таблица деталей заявки (товарные позиции)
-- =============================================
CREATE TABLE purchase_items (
    id INT PRIMARY KEY AUTO_INCREMENT,
    request_id INT NOT NULL,
    book_title VARCHAR(200) NOT NULL,
    author VARCHAR(100),
    isbn VARCHAR(20),
    year INT,
    publisher VARCHAR(100),
    quantity INT NOT NULL,
    FOREIGN KEY (request_id) REFERENCES purchase_requests(id) ON DELETE CASCADE
);
